--- 6_join_top_rated_per_category ---
SQL:
SELECT c.category_name, b.title, b.rating, b.price_gbp
        FROM books b
        JOIN categories c ON b.category_id = c.category_id
        WHERE b.rating >= 4
        ORDER BY c.category_name, b.rating DESC, b.price_gbp DESC
        LIMIT 15
